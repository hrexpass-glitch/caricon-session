function startSpeechRecognition() { const recognition = new (window.SpeechRecognition || window.webkitSpeechRecognition)(); recognition.continuous = true; // Enable continuous mode recognition.lang = 'ja-JP'; // Set language to Japanese recognition.interimResults = false; // Get only final results recognition.onstart = () => { console.log('Speech recognition started. Listening...'); }; recognition.onresult = (event) => { const transcript = event.results[event.resultIndex][0].transcript; console.log('Recognized speech:', transcript); // Handle the recognized speech }; recognition.onerror = (event) => { console.error('Speech recognition error:', event.error); // Handle errors appropriately if (event.error === 'no-speech') { console.warn('No speech was detected. Trying again...'); startSpeechRecognition(); // Restart recognition } }; recognition.onend = () => { console.log('Speech recognition ended. Restarting...'); startSpeechRecognition(); // Restart recognition }; try { recognition.start(); } catch (error) { console.error('Failed to start speech recognition:', error); } }
